Here's a minimal HTML racing game:

```html
<!DOCTYPE html>
<html>
<head>
<title>Racer</title>
<style>
body{margin:0;background:#222;display:flex;justify-content:center;align-items:center;height:100vh}
canvas{background:#333;border:2px solid #555}
</style>
</head>
<body>
<canvas id=c width=400 height=600></canvas>
<script>
const ctx=c.getContext('2d'),W=c.width,H=c.height;
let x=200,y=500,s=0,spd=5,obs=[],score=0;
document.onkeydown=e=>{
  if(e.key=='ArrowLeft')x-=10;
  if(e.key=='ArrowRight')x+=10;
  if(e.key=='ArrowUp')spd+=1;
  if(e.key=='ArrowDown')spd-=1;
  x=Math.max(50,Math.min(W-50,x));
  spd=Math.max(3,Math.min(10,spd));
};
function draw(){
  ctx.fillStyle='#333';ctx.fillRect(0,0,W,H);
  ctx.fillStyle='#555';ctx.fillRect(50,0,300,H);
  ctx.fillStyle='#777';
  for(let i=0;i<H;i+=40)ctx.fillRect(195,i,10,20);
  ctx.fillStyle='#f00';ctx.fillRect(x-20,y-30,40,60);
  ctx.fillStyle='#fff';ctx.fillRect(x-15,y-25,10,10);
  ctx.fillStyle='#fff';ctx.fillRect(x+5,y-25,10,10);
  obs.forEach(o=>{
    ctx.fillStyle='#aaa';ctx.fillRect(o.x,o.y,30,30);
    ctx.fillStyle='#666';ctx.fillRect(o.x+5,o.y+5,20,20);
  });
  ctx.fillStyle='#fff';ctx.font='16px sans-serif';
  ctx.fillText('Score: '+score,10,20);
  ctx.fillText('Speed: '+spd,10,40);
}
function update(){
  y-=spd;
  if(y<0){y=H-100;score+=100;}
  if(Math.random()<0.05)obs.push({x:Math.random()*300+50,y:0});
  obs.forEach(o=>{o.y+=spd;
    if(Math.abs(o.x-x)<30&&Math.abs(o.y-y)<30){
      score=Math.max(0,score-50);obs=[];y=H-100;
    }
  });
  obs=obs.filter(o=>o.y<H);
  draw();
  requestAnimationFrame(update);
}
update();
</script>
</body>
</html>
```

**Controls:**
- **← →** Steer left/right
- **↑ ↓** Speed up/slow down
- Avoid gray obstacles — hitting one resets your progress
- Score increases as you pass obstacles

**Features:**
- Road with dashed lane markings
- Red car with headlights
- Collision detection
- Score tracking
- Speed control
- Auto-scrolling track (when you reach the top, you respawn at the bottom)

Save as `racer.html` and open in a browser! 🏎️
