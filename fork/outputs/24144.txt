static void build_conversion_rule(const common_json & convert, std::string & out) {
    if (convert.contains("remove_prefix")) {
        std::string rule = convert["remove_prefix"].get<std::string>();
        out += rule + " ";
    }
    if (convert.contains("convert")) {
        std::string rule = convert["convert"].get<std::string>();
        if (rule == "int") {
            out += "integer";
        } else if (rule == "float") {
            out += "number";
        } else if (rule == "bool") {
            out += "boolean";
        } else {
            throw std::runtime_error("llama.cpp: unknown conversion type in grammar: " + rule);
        }
    }
}

static std::string escape_regex(std::string str) {
    // Don't escape characters inside character classes
    bool in_char_class = false;
    std::string result;
    for (size_t i = 0; i < str.size(); ++i) {
        if (str[i] == '[') {
            in_char_class = true;
            result += str[i];
        } else if (str[i] == ']') {
            in_char_class = false;
            result += str[i];
        } else if (!in_char_class && std::strchr(".^$|?()*+\\{}", str[i])) {
            result += '\\';
            result += str[i];
        } else {
            result += str[i];
        }
    }
    return result;
}
