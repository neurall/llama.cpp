Here is the smallest possible functional Breakout game in a single HTML file. It uses the HTML5 Canvas API and requires no external libraries.

```html
<!DOCTYPE html>
<html>
<body>
<canvas id="c" width="200" height="200"></canvas>
<script>
  const c = document.getElementById('c');
  const ctx = c.getContext('2d');
  let x = 100, y = 180, dx = 2, dy = -2;
  let bx = 80, by = 10, bw = 40, bh = 10;
  let bricks = [];
  for(let r=0; r<5; r++) for(let col=0; col<5; col++) bricks.push({x: col*40, y: r*10});

  function loop() {
    ctx.clearRect(0,0,200,200);
    ctx.fillStyle = '#000';
    ctx.fillRect(bx, by, bw, bh); // Paddle
    ctx.fillStyle = '#f00';
    ctx.fillRect(x, y, 5, 5); // Ball

    // Move Ball
    x += dx; y += dy;

    // Wall Collision
    if(x < 0 || x > 200) dx = -dx;
    if(y < 0 || y > 200) dy = -dy;

    // Paddle Collision
    if(y + 5 > by && y < by + bh && x > bx && x < bx + bw) dy = -dy;

    // Brick Collision
    bricks.forEach(b => {
      if(x > b.x && x < b.x + bw && y > b.y && y < b.y + bh) {
        dy = -dy;
        b.y = -100; // Remove brick
      }
    });

    requestAnimationFrame(loop);
  }
  loop();
</script>
</body>
</html>
```

1. **Canvas**: Creates a 200x200 drawing area.
2. **State**: Variables track the ball (`x, y, dx, dy`) and the paddle (`bx, by`).
