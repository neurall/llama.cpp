/* ---------- 7-bag randomizer ---------- */
function makeBag() {
  const keys = Object.keys(PIECES);
  const bag = [];
  for (let i = 0; i < keys.length; i++) bag.push(keys[i]);
  // shuffle
  for (let i = bag.length - 1; i > 0; i--) {
    const j = randInt(i + 1);
    [bag[i], bag[j]] = [bag[j], bag[i]];
  }
  return bag;
}

function createGrid() {
