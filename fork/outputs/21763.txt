// returns the length of the longest common prefix between two strings
size_t common_prefix_length(const std::string & a, const std::string & b) {
    size_t n = std::min(a.size(), b.size());
    size_t i = 0;
    while (i < n && a[i] == b[i]) {
        i++;
    }
    return i;
}

// returns the length of the longest common suffix between two strings
size_t common_suffix_length(const std::string & a, const std::string & b) {
    size_t n = std::min(a.size(), b.size());
    size_t i = 0;
    while (i < n && a[a.size() - 1 - i] == b[b.size() - 1 - i]) {
        i++;
    }
    return i;
}
