const P=[
  [[1,1,1,1]], // I
  [[1,1],[1,1]], // O
  [[0,1,0],[1,1,1]], // T
  [[1,0,0],[1,1,1]], // J
  [[0,0,1],[1,1,1]], // L
  [[0,1,1],[1,1,0]], // S
  [[1,1,0],[0,1,1]]  // Z
];
const C=['#0ff','#ff0','#f0f','#00f','#f80','#0f0','#f00'];

function newPiece(){
  pT=Math.floor(Math.random()*7);
  p=P[pT];
  pX=Math.floor(W/2-p[0].length/2);
  pY=0;pR=0;
}

function rotate(){
  let r=[];
  for(let i=0;i<p[0].length;i++){
    r[i]=[];
    for(let j=0;j<p.length;j++){
      r[i][j]=p[p.length-1-j][i];
    }
  }
  p=r;
}

function collide(){
  for(let i=0;i<p.length;i++){
    for(let j=0;j<p[i].length;j++){
      if(p[i][j]){
        let nx=pX+j,ny=pY+i;
        if(nx<0||nx>=W||ny>=H)return true;
        if(ny>=0&&b[ny]&&b[ny][nx])return true;
      }
    }
  }
  return false;
}

function merge(){
  for(let i=0;i<p.length;i++){
    for(let j=0;j<p[i].length;j++){
      if(p[i][j]){
        let ny=pY+i,nx=pX+j;
        if(ny>=0){
          if(!b[ny])b[ny]=[];
          b[ny][nx]=pT+1;
        }
      }
    }
  }
}

  for(let i=H-1;i>=0;i--){
    if(b[i]&&b[i].every(v=>v)){
      b.splice(i,1);
      s+=100;
      i++;
    }
  }
}

  x.fillStyle='#000';
