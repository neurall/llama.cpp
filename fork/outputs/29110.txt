static bool is_model_url(const std::string & url) {
    return url.rfind("hf://", 0) == 0 || url.rfind("ms://", 0) == 0;
}

static bool is_model_url(const common_params_model & model) {
    return is_model_url(model.url);
}

static bool is_model_url(const common_params & params) {
    return is_model_url(params.model) || is_model_url(params.model.mmproj) || is_model_url(params.model.mtp);
}

static bool is_model_url(const common_params_speculative & spec) {
    return is_model_url(spec.draft) || is_model_url(spec.mtp);
}

static bool is_model_url(const common_params & params, const common_params & params_base) {
    return is_model_url(params.model) || is_model_url(params.model) != is_model_url(params_base.model) ||
        is_model_url(params.model.mmproj) || is_model_url(params.model.mmproj) != is_model_url(params_base.model.mmproj) ||
        is_model_url(params.model.mtp) || is_model_url(params.model.mtp) != is_model_url(params_base.model.mtp) ||
        (is_model_url(params.speculative) && !spec_types_is_default(params_base.speculative));
}

static bool is_model_url(const common_params & params, const common_params & params_base, const std::string & hf_repo, const std::string & hf_file, const std::string & ms_repo, const std::string & ms_file) {
    return !hf_repo.empty() || !ms_repo.empty() || !is_model_url(params, params_base);
}
