bool parse_cpu_mask(const std::string & mask, std::vector<int32_t> & cpumask) {
    cpumask.clear();
    if (mask.empty()) {
        return true;
    }
    std::string hex = mask;
    if (hex.size() > 1 && hex[0] == '0' && hex[1] == 'x') {
        hex = hex.substr(2);
    }
