function makeBag() {
  const bag = [];
  for (let i = 0; i < PIECES.length; i++) bag.push(i);
  // shuffle
  for (let i = bag.length - 1; i > 0; i--) {
    const j = randInt(i + 1);
    [bag[i], bag[j]] = [bag[j], bag[i]];
  }
  return bag;
}
