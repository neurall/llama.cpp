#include <stdlib.h>
#include <time.h>

int main() {
    initscr();
    hidecursor();
    srand(time(0));
    
    int x = 10, y = 10, dx = 1, dy = 0;
    int px = x, py = y; // head
    int len = 1;
    int fx = rand() % 20 + 1, fy = rand() % 20 + 1; // food
    
    while(1) {
        int c = getch();
        if(c == 97) dx = -1; // a
        if(c == 100) dx = 1; // d
        if(c == 104) dy = -1; // h
        if(c == 108) dy = 1; // l
        
        x += dx; y += dy;
        
        if(x < 0 || x > 20 || y < 0 || y > 20) break;
        
        // Move tail
        for(int i = 0; i < len; i++) {
            int tx = px, ty = py;
            px = x; py = y;
            x = tx; y = ty;
        }
