const c = document.getElementById('c');
const ctx = c.getContext('2d');
let x = 100, y = 150, dx = 2, dy = -2, bx = 100, by = 180;
let bricks = [];
for(let r=0; r<5; r++) for(let col=0; col<10; col++) bricks.push({x:col*20, y:r*20, alive:1});

function loop() {
    ctx.clearRect(0,0,200,200);
    ctx.fillStyle = '#000';
    ctx.fillRect(bx, by, 20, 10); // Paddle
    
    // Move Ball
    x += dx; y += dy;
    if(x < 0 || x > 200) dx = -dx;
    if(y < 0 || y > 200) dy = -dy;
    
    // Paddle Collision
    if(y > by && x > bx && x < bx+20) dy = -dy;
    
    // Brick Collision
    bricks.forEach(b => {
        if(b.alive && x > b.x && x < b.x+20 && y > b.y && y < b.y+20) {
            b.alive = 0; dy = -dy;
        }
    });
    
    // Draw Bricks
    ctx.fillStyle = '#f00';
    bricks.forEach(b => {
        if(b.alive) ctx.fillRect(b.x, b.y, 20, 10);
    });
    
    // Draw Ball
    ctx.fillStyle = '#fff';
    ctx.fillRect(x, y, 5, 5);
    
    requestAnimationFrame(loop);
}
