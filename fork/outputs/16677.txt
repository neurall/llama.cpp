static int count_lines(const std::string & filename) {
    std::ifstream file(filename);
    if (!file.is_open()) {
        return -1;
    }
    int line_count = 0;
    std::string line;
    while (std::getline(file, line)) {
        ++line_count;
    }
    return line_count;
}

}
