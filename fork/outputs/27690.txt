const c = document.getElementById('c');
const ctx = c.getContext('2d');
let x = 100, y = 180, dx = 2, dy = -2, bx = 100, by = 100;
let bricks = [];
for(let i=0; i<10; i++) bricks.push({x: i*20, y: 20});

function loop() {
    ctx.clearRect(0,0,200,200);
    ctx.fillStyle = '#000';
    ctx.fillRect(bx-10, by-10, 20, 20); // Paddle
    
    // Move Paddle
    bx += dx;
    if(bx < 10) bx = 10;
    if(bx > 190) bx = 190;

    // Move Ball
    by += dy;
    if(by < 10) dy = -dy; // Top bounce
    if(by > 190) { // Game Over
        x = 100; y = 180; dx = 2; dy = -2; bx = 100; by = 100;
    }

    // Brick Collision
    bricks.forEach(b => {
        if(x > b.x && x < b.x+20 && y > b.y && y < b.y+20) {
            dy = -dy;
            bricks = bricks.filter(br => br !== b);
        }
    });

    // Draw Ball
    ctx.fillStyle = '#fff';
    ctx.fillRect(x-5, y-5, 10, 10);
    
    // Draw Bricks
    ctx.fillStyle = '#f00';
    bricks.forEach(b => ctx.fillRect(b.x, b.y, 20, 20));

    requestAnimationFrame(loop);
}
