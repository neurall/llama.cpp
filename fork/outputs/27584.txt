#include <vector>
#include <random>
#include <chrono>
#include <thread>

using namespace std;

int main() {
    // Grid size
    const int W = 20, H = 15;
    vector<int> snake(W * H, 0); // 0 = empty, 1 = snake, 2 = food
    int head = W * H / 2; // Start in middle
    int dir = 0; // 0=right, 1=down, 2=left, 3=up
    int score = 0;

    // Place food
    int food = rand() % (W * H);
    snake[food] = 2;

    auto move = [&](int d) {
        dir = d;
    };

    while (true) {

        char c = getchar();
        if (c == 'w') move(3);
        else if (c == 's') move(1);
        else if (c == 'a') move(2);
        else if (c == 'd') move(0);
        else if (c == 'q') break;

        // Calculate new head position
        int nx = head % W, ny = head / W;

        else if (dir == 1) ny++;
        else if (dir == 2) nx--;
        else if (dir == 3) ny--;

        if (nx < 0 || nx >= W || ny < 0 || ny >= H) break;
