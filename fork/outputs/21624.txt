bool parse_cpu_mask(const std::string & mask, std::vector<int32_t> & cpumask) {
    if (mask.empty()) {
        return false;
    }
    if (mask.size() > 64) {
        throw std::invalid_argument("mask too long");
    }
    cpumask.clear();
    for (size_t i = 0; i < mask.size(); i++) {
        char c = mask[i];
        if (c == '0' || c == '1') {
            cpumask.push_back(c - '0');
        } else {
            throw std::invalid_argument("invalid mask");
        }
    }
    return true;
}
