```cpp
static void print_line_count(const std::string & path = "common/json-schema-to-grammar.cpp") {
    std::ifstream file(path);
    if (!file.is_open()) {
        std::cerr << "Error: could not open file: " << path << std::endl;
        return;
    }

    int line_count = 0;
    std::string line;
    while (std::getline(file, line)) {
        ++line_count;
    }

    std::cout << path << " has " << line_count << " lines." << std::endl;
}
```

