           self.value = value
           self.prev = None

   class LRUCache:
       def __init__(self, capacity: int):
           self.capacity = capacity
           self.cache = {}
           self.head = Node(0, 0)
           self.tail = Node(0, 0)
           self.head.next = self.tail
           self.tail.prev = self.head

       def _add_to_end(self, node):
           prev = self.tail.prev
           prev.next = node
           node.prev = prev
           node.next = self.tail
           self.tail.prev = node

       def _remove(self, node):
           prev = node.prev
           nxt = node.next
           prev.next = nxt
           nxt.prev = prev

       def put(self, key: int, value: int) -> None:

               node = self.cache[key]
               node.value = value
               self._remove(node)
