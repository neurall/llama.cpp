#include <iostream>
#include <vector>
#include <random>
#include <chrono>

using namespace std;

const int W = 10, H = 20;
int board[H][W] = {0};
int score = 0;

// Tetromino shapes (7 types)
int shapes[7][4][2] = {
    {{0,0},{0,1},{0,2},{0,3}}, // I
    {{0,0},{0,1},{1,0},{1,1}}, // O
    {{0,0},{0,1},{0,2},{1,2}}, // T
    {{0,0},{0,1},{1,1},{2,1}}, // J
    {{0,1},{0,0},{1,0},{2,0}}, // L
    {{0,0},{0,1},{1,0},{1,1}}, // Z (same as O but different logic in real tetris, simplified here)
    {{0,0},{0,1},{1,1},{1,2}}  // S
};

int currentShape, currentX, currentY, rotation;

void draw() {
    for (int y = 0; y < H; ++y) {
        for (int x = 0; x < W; ++x) {
            if (board[y][x]) cout << "█";
            else {
                // Check if current piece is here
                bool isPiece = false;
                for (auto p : shapes[currentShape]) {
                    if (p[0] + currentX == x && p[1] + currentY == y) {
                        isPiece = true;
                        break;
                    }
                }
                cout << (isPiece ? "■" : " ");
            }
        }
        cout << "\n";
    }
