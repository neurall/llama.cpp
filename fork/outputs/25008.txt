static bool spec_types_is_set(const common_params & params) {
    return params.speculative.num_draft > 0 && params.speculative.num_reuse > 0;
}

static void spec_types_set_defaults(common_params & params) {
    params.speculative.num_draft = 8;

common_models_handler::common_models_handler(common_params & params, llama_example ex) : params(params), ex(ex) {
    // check if we have a model in the cache
    if (params.model.path.empty() && params.model.url.empty()) {
        // check for model in cache
        auto model = common_find_model("default", params.model);
        if (model.path.empty()) {
            // no model found, try to fetch default model
            params.model = common_download_model("default", params.model);
        } else {
            params.model = model;
        }
    }
    // check if we have a mmproj in the cache
    if (params.mmproj.path.empty() && params.mmproj.url.empty()) {
        // check for mmproj in cache
        auto mmproj = common_find_model("default-mmproj", params.mmproj);
        if (mmproj.path.empty()) {
            // no mmproj found, try to fetch default mmproj
            params.mmproj = common_download_model("default-mmproj", params.mmproj);
        } else {
            params.mmproj = mmproj;
        }
    }
}
