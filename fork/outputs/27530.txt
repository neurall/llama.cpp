#include <random>
#include <chrono>

using namespace std;

int main() {
    // Terminal dimensions
    const int W = 20, H = 15;

    vector<pair<int,int>> snake = {{1,1}};
    pair<int,int> food = {5,5};
    int dir = 0; // 0:up, 1:right, 2:down, 3:left
    int score = 0;
    
    // Random seed for food placement

                cout << " ";

        }
    };

        // Check collision
