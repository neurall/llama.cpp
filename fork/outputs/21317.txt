   const S=20, W=10, H=10; // cell size, grid width/height
   let snake=[{x:5,y:5}], dir={x:1,y:0}, food={x:7,y:7}, score=0;
   function draw(){
     ctx.fillStyle='#000'; ctx.fillRect(0,0,c.width,c.height);
     ctx.fillStyle='#0f0';

   function move(){
     let nx=snake[0].x+dir.x, ny=snake[0].y+dir.y;
