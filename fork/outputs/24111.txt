static std::string format_regex(const std::string & s) {
    // Convert a regex to GBNF grammar format
    std::string result;
    bool in_char_class = false;
    bool escaped = false;

    for (size_t i = 0; i < s.size(); ++i) {
        char c = s[i];

        if (escaped) {
            result += c;
            escaped = false;
            continue;
        }

        if (c == '\\') {
            result += '\\';
            escaped = true;
            continue;
        }

        if (c == '[') {
            in_char_class = true;
            result += '[';
        } else if (c == ']') {
            in_char_class = false;
            result += ']';
        } else if (c == '"' && !in_char_class) {
            result += "\\\"";
        } else if (c == '|' && !in_char_class) {
            result += " | ";
        } else if (c == '(' && !in_char_class) {
            result += "(";
        } else if (c == ')' && !in_char_class) {
            result += ")";
        } else if (c == '{' && !in_char_class) {
            result += "{";
        } else if (c == '}' && !in_char_class) {
            result += "}";
        } else if (c == '*' && !in_char_class) {
            result += "*";
        } else if (c == '+' && !in_char_class) {
            result += "+";
        } else if (c == '?' && !in_char_class) {
            result += "?";
        } else if (c == '.' && !in_char_class) {
            result += "[^\"\\\\ \\x7f \\x00-\\x1f]";
        } else {
            result += c;
        }
    }

    return result;
}

static std::string build_grammar(const std::function<void(const common_grammar_builder &)> & cb, const std::vector<std::string> & hacks) {
    auto t0 = ggml_time_us();
    auto bnf = common_grammar_builder_create();

    cb(bnf);
    for (auto & hack : hacks) {
        if (hack == "space") {
            common_grammar_builder_add_rule(bnf, "space", "( \" \" | \"\\n\"{1,2} [ \\t]{0,20})?");
        }
    }

    auto t1 = ggml_time_us();
    LOG_INF("%.2f ms to build grammar\n", (t1 - t0) / 1000.0f);

    std::stringstream out;
    common_grammar_builder_write(bnf, out);

    auto t2 = ggml_time_us();
    LOG_INF("%.2f ms to write grammar\n", (t2 - t1) / 1000.0f);

    common_grammar_builder_free(bnf);
    return out.str();
}
