static bool is_default_model(const common_params_model & model) {
    return model.path.empty() && model.url.empty() && model.hf_repo.empty();
}

static bool is_default_mmproj(const common_params_model & mmproj) {
    return is_default_model(mmproj) && mmproj.auto_download == MMFILE_AUTO;
}

static bool is_default_mtp(const common_params_model & mtp) {
    return is_default_model(mtp) && mtp.auto_download == MMFILE_NONE;
}

static bool is_default_preset(const std::string & preset) {
    return preset.empty();
}

static bool is_default_model_config(const common_params_model_config & config) {
    return config.vocab_bpe.empty() && config.tokensets.empty() && config.add_bos_token == -1;
}

static bool is_default_model_path(const common_params & params) {
    return is_default_model(params.model) && is_default_mmproj(params.mmproj) && is_default_mtp(params.mtp) && is_default_preset(params.model_config.preset) && params.model_config.lora.empty();
}

static bool is_default_model_url(const common_params & params) {
    return params.model.url.empty() && params.mmproj.url.empty() && params.mtp.url.empty() && is_default_preset(params.model_config.preset) && params.model_config.lora.empty();
}

static bool is_default_hf_repo(const common_params & params) {
    return params.model.hf_repo.empty() && params.mmproj.hf_repo.empty() && params.mtp.hf_repo.empty() && is_default_preset(params.model_config.preset) && params.model_config.lora.empty();
}

static bool is_default_model_path_or_url(const common_params & params) {
    return is_default_model_path(params) && is_default_model_url(params);
}

static bool is_default_model_path_or_hf_repo(const common_params & params) {
    return is_default_model_path(params) && is_default_hf_repo(params);
}

static bool is_default_model_url_or_hf_repo(const common_params & params) {
    return is_default_model_url(params) && is_default_hf_repo(params);
}
