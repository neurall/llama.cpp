static std::string build_repetition(const std::string & item_rule, int min_items, int max_items, const std::string & separator_rule = "") {

    if (min_items == 0 && max_items == 1) {
        return item_rule + "?";

            return item_rule + "+";

            return item_rule + "*";
        }

}
