static void handle_model_result(const common_params & params, const std::string & model_url, const std::string & model_filename, handle_model_result & result) {
    // handle preset
    if (string_ends_with(model_filename, ".json")) {
        result.preset_path = model_filename;
        result.found_preset = true;
        return;
    }

    // handle mtp
    if (string_ends_with(model_filename, ".mtp")) {
        result.mtp = {model_filename, model_filename};
        result.found_mtp = true;
        return;
    }

    // handle mmproj
    if (string_ends_with(model_filename, ".mmproj")) {
        result.mmproj = {model_filename, model_filename};
        result.found_mmproj = true;
        return;
    }

    // handle model
    if (params.model.empty()) {
        params.model = {model_url, model_filename};
    }

    // handle speculative model
    if (params.speculative.model.empty() && params.speculative.n_draft_max > 0 && !spec_types_is_default(params)) {
        params.speculative.model = {model_url, model_filename};
    }
}
