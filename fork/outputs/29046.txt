static bool is_download(const common_params_model & model) {
    return is_url(model.url) || !model.hf_repo.empty();
}

static common_params_model handle_hf_download(
    common_params & params,
    const std::string & hf_model,
    const std::string & hf_file,
    const std::string & hf_repo,
    const std::string & hf_repo_type,
    const std::string & hf_token,
    const std::string & local_path,
    const std::string & mmproj) {
    common_params_model result;
    result.hf_model = hf_model;
    result.hf_file = hf_file;
    result.hf_repo = hf_repo.empty() ? "ggml-org/models" : hf_repo;
    result.hf_repo_type = hf_repo_type.empty() ? "model" : hf_repo_type;
    result.hf_token = hf_token;
    result.path = local_path.empty() ? fs_get_cache_file(hf_model) : local_path;
    if (!mmproj.empty()) {
        result.mmproj.path = mmproj;
    }
    params.download_cache = fs_get_cache_file(clean_file_name(hf_model));
    return result;
}

static common_params_model handle_url_download(
    common_params & params,
    const std::string & url,
    const std::string & local_path,
    const std::string & mmproj) {
    common_params_model result;
    result.url = url;
    result.path = local_path.empty() ? get_default_local_path(url) : local_path;
    if (!mmproj.empty()) {
        result.mmproj.path = mmproj;
    }
    params.download_cache = fs_get_cache_file(clean_file_name(url));
    return result;
}
