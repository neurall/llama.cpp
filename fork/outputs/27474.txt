Here is the smallest possible functional Breakout game in a single HTML file. It uses the HTML5 Canvas API and requires no external libraries.

```html
<!DOCTYPE html>
<html>
<body>
<canvas id="c" width="200" height="200"></canvas>
<script>
const c = document.getElementById('c');
const ctx = c.getContext('2d');
let x = 100, y = 180, dx = 2, dy = -2, bx = 100, by = 10;
let p = 80, pw = 20;
let bricks = [];
for(let i=0; i<10; i++) for(let j=0; j<5; j++) bricks.push({x: 10+i*20, y: 10+j*20, alive: true});

function draw() {
  ctx.clearRect(0,0,200,200);
  ctx.fillStyle = '#000';
  ctx.fillRect(p, 190, pw, 10);
  ctx.fillRect(x, y, 5, 5);
  bricks.forEach(b => {
    if(b.alive) {
      ctx.fillRect(b.x, b.y, 18, 10);
      if(x > b.x && x < b.x+18 && y > b.y && y < b.y+10) {
        dy = -dy; b.alive = false;
      }
    }
  });
  if(y <= 0 || y >= 190) dy = -dy;
  if(x < p || x > p+pw) dx = -dx;
  x += dx; y += dy;
  requestAnimationFrame(draw);
}
document.addEventListener('keydown', e => {
  if(e.key === 'ArrowLeft') p -= 10;
  if(e.key === 'ArrowRight') p += 10;
});
draw();
</script>
</body>
</html>
```

### How it works:
1. **Canvas**: A 200x200 pixel canvas.
2. **Ball**: A 5x5 square at `(x, y)` moving with velocity `(dx, dy)`.
