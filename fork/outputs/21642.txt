bool parse_cpu_mask(const std::string & mask, std::vector<int32_t> & cpumask) {
    if (mask.empty()) {
        return false;
    }
    cpumask.clear();
    for (size_t i = 0; i < mask.size(); i++) {
        if (mask[i] == '0' || mask[i] == '1') {
            cpumask.push_back(mask[i] - '0');
        } else {
            return false;
        }
    }
    return true;
}
