common_models_handler::common_models_handler(common_params & params, llama_example ex) : params(params), ex(ex) {
    // check if we have a model in the cache
    if (params.model.path.empty() && params.model.url.empty()) {
        // check for model in cache
        auto model_cache = fs_get_cache_list("models");
        if (!model_cache.empty()) {
            // find model with the highest modification time
            std::string best_model;
            std::time_t best_time = 0;
            for (const auto & model : model_cache) {
                auto ftime = fs_get_mtime(model);
                if (ftime > best_time) {
                    best_time = ftime;
                    best_model = model;
                }
            }
            if (!best_model.empty()) {
                params.model.path = best_model;
                LOG_INF("%s: using cached model: %s\n", __func__, params.model.path.c_str());
            }
        }
    }
}

common_models_handler::~common_models_handler() {
    if (params.model.path.empty() && params.model.url.empty()) {
        throw std::runtime_error("error: no model path or URL was specified");
    }
}

static std::string get_model_url(const std::string & model) {
    if (string_starts_with(model, "hf://")) {
        return model;
    }
    if (string_starts_with(model, "gguf://")) {
        return model;
    }
    if (string_starts_with(model, "http://") || string_starts_with(model, "https://")) {
        return model;
    }
    return "hf://" + model;
}
