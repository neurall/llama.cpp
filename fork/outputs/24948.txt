static bool spec_types_is_default(const common_params & params) {

        && params.speculative.num_draft_min == 1
        && params.speculative.num_gpu == 999
        && params.speculative.accept_grammar == ""
        && params.speculative.accept_per_position == ""
        && params.speculative.accept_per_draft == ""
        && params.speculative.accept_per_position_max == 0
        && params.speculative.accept_per_draft_max == 0;
}
