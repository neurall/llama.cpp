static void handle_model_download(const std::string & value) {
    common_params params;
    common_params_parser_init(params, LLAMA_EXAMPLE_COMMON, nullptr);
    params.model.url = value;
    common_download_params dl_params;
    dl_params.hf_repo = params.model.hf_repo;
    dl_params.hf_file = params.model.hf_file;
    common_download(params.model.url, dl_params, true);
    exit(0);
}

static void handle_model_pretrained_type(common_params & params, const std::string & value) {
    if (value == "none") {
        params.model.pretrained_type = COMMON_PRETRAINED_TYPE_BASE;
    } else if (value == "mean") {
        params.model.pretrained_type = COMMON_PRETRAINED_TYPE_MEAN;
    } else {
        throw std::invalid_argument("invalid --" + std::string(LLAMA_ARG_MODEL_PRETRAINED_TYPE) + " value, expected: 'none' or 'base', 'mean'");
    }
}
