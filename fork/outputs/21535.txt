import csv
from collections import defaultdict

def process_sales_csv(filename):
    """
    Reads a CSV file with columns: date, region, amount.
    Prints total amount per region and per month.
    """
    # Initialize dictionaries to store totals
    region_totals = defaultdict(float)
    month_totals = defaultdict(float)

    try:
        with open(filename, 'r', newline='') as f:
            reader = csv.DictReader(f)
            for row in reader:
                # Parse date to extract month (YYYY-MM format)
                date_str = row['date'].strip()
                # Extract month as YYYY-MM
                month = date_str[:7] if len(date_str) >= 7 else date_str

                region = row['region'].strip()
                amount = float(row['amount'].strip())

                # Accumulate totals
                region_totals[region] += amount
                month_totals[month] += amount

    except FileNotFoundError:
        print(f"Error: File '{filename}' not found.")
        return
    except KeyError as e:
        print(f"Error: Missing column {e} in CSV.")
        return
    except ValueError as e:
        print(f"Error: Invalid data format. {e}")
        return

    # Print totals per region
    print("Total amount per region:")
    for region, total in sorted(region_totals.items()):
        print(f"  {region}: {total:.2f}")

    # Print totals per month
    print("\nTotal amount per month:")
    for month, total in sorted(month_totals.items()):
        print(f"  {month}: {total:.2f}")

if __name__ == "__main__":
    # Example usage: replace 'sales.csv' with your actual file path
    process_sales_csv('sales.csv')
