### Requirements
- C++11 or later
- Terminal with support for ANSI escape codes (most modern terminals)

#include <vector>
#include <random>
#include <chrono>

using namespace std;

int main() {
    // Terminal dimensions
    const int W = 20, H = 15;
    
    // Snake state: head at (1,1), body at (1,1) initially
    vector<pair<int,int>> snake = {{1,1}};
    pair<int,int> dir = {0,1}; // Direction: right

    auto rand_pos = [&]() {
        return {rand() % (W-2) + 1, rand() % (H-2) + 1};
    };

    cout << "\033[2J"; // Clear screen
    cout << "\033[H";  // Move cursor to top-left

                    else if(is_food) cout << "F";
                    else cout << " ";

            cout << "\n";
