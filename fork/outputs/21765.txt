int32_t common_cpu_get_num_cores() {
    int32_t num_cores = common_cpu_get_num_physical_cores();

// returns true if the string contains only ASCII printable characters
bool string_is_ascii(const std::string & str) {
    return std::all_of(str.begin(), str.end(), [](unsigned char c) {
        return c <= 0x7F;
    });
}
