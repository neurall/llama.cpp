```python
import asyncio
import aiohttp
import random

# Define a function to handle retries
async def download_page(session, url, retries=3):
    delay = 2 ** retries  # Start with a small delay and double it each time
    while retries > 0:
        try:
            async with session.get(url) as response:
                if response.status == 200:
                    # Successful download, save the page
                    with open(f"{url.split('/')[-1]}", "wb") as f:
                        f.write(await response.content())
                    return True
                else:
                    print(f"Failed to download {url} after {delay} attempts.")
                    retries -= 1
                    await asyncio.sleep(delay)
                    continue
        except Exception as e:
            print(f"Error downloading {url}: {e}")
            retries -= 1
            await asyncio.sleep(delay)
    return False  # If we reach here, the loop should exit

# Define the main async function
async def main():
    urls = ["https://example.com/page{i}".format(i=i) for i in range(1, 11)]  # Example URLs
    session = aiohttp.ClientSession()

    # Define the coroutine to download pages
    async def download_pages(session, urls):
        tasks = []
        for url in urls:
            tasks.append(download_page(session, url))
        await asyncio.gather(*tasks)

    # Start the download process
    await download_pages(session, urls)

# Run the main function
asyncio.run(main())
```
