struct common_chat_templates_ptr {
    common_chat_templates * tmpls = nullptr;
    bool                    has_explicit_template = false;

    common_chat_templates_ptr() = default;
    common_chat_templates_ptr(common_chat_templates * tmpls) : tmpls(tmpls) {}
    common_chat_templates_ptr(common_chat_templates * tmpls, bool has_explicit_template)
        : tmpls(tmpls), has_explicit_template(has_explicit_template) {}

    ~common_chat_templates_ptr() { delete tmpls; }

    common_chat_templates_ptr(const common_chat_templates_ptr & other) = delete;
    common_chat_templates_ptr & operator=(const common_chat_templates_ptr & other) = delete;

    operator bool() const { return tmpls != nullptr; }

    common_chat_templates * get() const { return tmpls; }
    common_chat_templates * release() {
        auto ptr = tmpls;
        tmpls = nullptr;
        return ptr;
    }
};
