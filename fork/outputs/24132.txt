static std::string escape_json_string(const std::string & input) {
    std::string result = "\"";
    for (auto c : input) {
        if (c == '\n') {
            result += "\\n";
        } else if (c == '\t') {
            result += "\\t";
        } else if (c == '\"') {
            result += "\\\"";
        } else if (c == '\\') {
            result += "\\\\";
        } else {
            result += c;
        }
    }
    result += "\"";
    return result;
}

static std::string add_whitespace(std::string grammar) {
    // add whitespace to make the grammar more readable
    static const std::regex substitution_pattern(R"(~==)");
    std::sregex_token_iterator token_begin(grammar.begin(), grammar.end(), substitution_pattern, -1), token_end{};
    std::string result;
    for (auto it = token_begin; it != token_end; ++it) {
        result += regex_replace(*it, std::regex(R"(^(.*?):)"), "$1\n  ");
        result += regex_replace(*it, std::regex(R"(^(.+?) \| )"), "$1  | ");
    }
    return result + ")";
}
