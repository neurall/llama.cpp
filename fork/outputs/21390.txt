Here is the smallest possible functional Snake game in a single HTML file (approx. 400 bytes). It uses a grid-based approach with a single `<canvas>` element.

```html
<!DOCTYPE html>
<html>
<head><title>Snake</title></head>
<body>
<canvas id="c" width="200" height="200"></canvas>
<script>
  const c = document.getElementById('c');
  const ctx = c.getContext('2d');
  const w = 20, h = 20;
  let x = 10, y = 10, dx = 1, dy = 0, food = {x: 15, y: 15};
  
  function draw() {
    ctx.fillStyle = '#000';
    ctx.fillRect(0, 0, w * 10, h * 10);
    ctx.fillStyle = '#0f0';
    ctx.fillRect(x * 10, y * 10, 10, 10);
    ctx.fillStyle = '#f00';
    ctx.fillRect(food.x * 10, food.y * 10, 10, 10);
  }

  function update() {
    x += dx;
    y += dy;
    if (x < 0 || x >= w || y < 0 || y >= h) {
      x = 10; y = 10; dx = 1; dy = 0;
    }
    if (x === food.x && y === food.y) {
      food.x = Math.floor(Math.random() * w);
      food.y = Math.floor(Math.random() * h);
    }
    draw();
  }

  document.addEventListener('keydown', e => {
    if (e.key === 'ArrowUp') dy = -1;
    if (e.key === 'ArrowDown') dy = 1;
    if (e.key === 'ArrowLeft') dx = -1;
    if (e.key === 'ArrowRight') dx = 1;
  });

  setInterval(update, 100);
</script>
</body>
</html>
```

### Features:
- **Controls**: Arrow keys.
- **Gameplay**: Snake
