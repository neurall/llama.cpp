static std::unordered_map<std::string, BuiltinRule> PREFIX_RULES = {
    {"bool", {"boolean"}},
    {"decimal", {"decimal-part"}},
    {"integer", {"integral-part"}},
    {"number", {"number"}},
    {"value", {"value"}},
    {"object", {"object"}},
    {"array", {"array"}},
    {"uuid", {"uuid"}}
};

static std::string jsonSchemaToGrammar(const json& json) {
    std::string grammar = "{\n";

    for (const auto& rule : json) {
        if (rule.first.find('$') != std::string::npos) {
            continue; // Skip external grammars
        }

        auto [json_schema_rule, dependencies] = rule;
        grammar += fst::LexicalGrammar::escape(json_schema_rule) + ";" + SPACE_RULE;

        if (dependencies.size()) {
            grammar += " " + PREFIX_RULES[json_schema_rule].content + " " + join(", ", dependencies) + ";" + "\n";
        } else {
            grammar += json_schema_rule + ";" + "\n";
        }
    }

    grammar += "}";
    return grammar;
}
