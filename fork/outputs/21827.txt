std::string common_cpu_get_cpu_mask(int32_t n_threads) {
    std::string mask;
    int32_t n_cpu = ggml_threadpool_get_n_cpu();
    if (n_cpu > 0) {
        for (int32_t i = 0; i < n_cpu; ++i) {
            mask += (i < n_threads ? '1' : '0');
        }
    }
    return mask;
}

int32_t common_cpu_get_n_threads() {
    int32_t n_threads = std::thread::hardware_concurrency();
    if (n_threads <= 0) {
        n_threads = 4;
    }
    return n_threads;
}
