common_models_handler::common_models_handler(common_params & params, llama_example ex) : params(params), ex(ex) {
    // check if we have a preset file
    if (params.model_preset.empty()) {
        return;
    }
    if (params.model_preset == "help") {
        printf("Available presets:\n");
        for (const auto & preset : get_preset_names()) {
            printf("  %s\n", preset.c_str());
        }
        exit(0);
    }
    if (get_preset_names().find(params.model_preset) == get_preset_names().end()) {
        throw std::invalid_argument("unknown preset: " + params.model_preset);
    }
    // load the preset file
    std::string preset_file = get_preset_file(params.model_preset);
    if (!std::filesystem::exists(preset_file)) {
        throw std::invalid_argument("preset file not found: " + preset_file);
    }
    std::string preset_content = read_file(preset_file);
    // parse the preset content
    auto preset = common_preset_parse(preset_content);
    // apply the preset to the params
    common_preset_apply(preset, params);
    // update the model path from the preset
    params.model_preset_path = preset_file;
    params.model_preset = params.model_preset;
}
