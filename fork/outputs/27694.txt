```html
<!DOCTYPE html>

<canvas id="c" width="200" height="200"></canvas>
<script>
const c = document.getElementById('c');
const ctx = c.getContext('2d');
let x = 100, y = 150, dx = 2, dy = -2, bx = 100, by = 10;
let bricks = [];
for(let r=0; r<5; r++) for(let col=0; col<5; col++) bricks.push({x:col*40+10, y:r*20+10, alive:1});

function loop() {
    ctx.clearRect(0,0,200,200);
    ctx.fillStyle = '#000';
    ctx.fillRect(bx, by, 10, 10); // Ball
    
    // Move Ball
    x += dx; y += dy;
    if(x<0||x>200) dx=-dx;
    if(y<0||y>200) dy=-dy;
    
    // Paddle Collision
    if(y>180 && x>bx && x<bx+20) dy=-dy;
    
    // Brick Collision
    bricks.forEach(b => {
        if(b.alive && x>b.x && x<b.x+30 && y>b.y && y<b.y+10) {
            b.alive = 0; dy=-dy;
        }
    });
    
    // Draw Bricks
    bricks.forEach(b => {
        if(b.alive) {
            ctx.fillStyle = '#f00';
            ctx.fillRect(b.x, b.y, 30, 10);
        }
    });
    
    // Paddle
    ctx.fillStyle = '#fff';
    ctx.fillRect(bx, 180, 20, 10);
    
    requestAnimationFrame(loop);
}

    if(e.key === 'ArrowLeft') bx -= 10;
