Here is the smallest possible functional Breakout game in a single HTML file. It uses the HTML5 Canvas API and requires no external libraries.

```html
<!DOCTYPE html>
<html>
<body>
<canvas id="c" width="200" height="200"></canvas>
<script>
const c = document.getElementById('c');
const ctx = c.getContext('2d');
let x = 100, y = 180, dx = 2, dy = -2, bx = 100, by = 100;
let bricks = [];
for(let i=0; i<10; i++) bricks.push({x: i*20, y: 20});

function loop() {
    ctx.fillStyle = '#000'; ctx.fillRect(0,0,200,200);
    ctx.fillStyle = '#fff';
    ctx.fillRect(bx-10, by-10, 20, 20); // Paddle
    ctx.fillRect(x-5, y-5, 10, 10);    // Ball

    // Collision with bricks
    bricks.forEach(b => {
        if(x > b.x && x < b.x+20 && y > b.y && y < b.y+20) {
            dy = -dy;
            bricks = bricks.filter(br => br !== b);
        }
    });

    // Wall collision
    if(x <= 0 || x >= 200) dx = -dx;
    if(y <= 0 || y >= 200) dy = -dy;

    // Paddle collision
    if(y >= by-10 && y <= by+10 && x >= bx-10 && x <= bx+10) dy = -dy;

    x += dx; y += dy;
    requestAnimationFrame(loop);
}
loop();
</script>
</body>
</html>
```

### How to play:
1. Save the code as `breakout.html`.
2. Open it in any browser.
3. The paddle (white square at the bottom) moves automatically to catch the ball.
4. Hit the bricks to destroy them.
5. If the ball hits the bottom, the game resets (loop continues).
