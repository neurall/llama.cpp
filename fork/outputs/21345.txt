Here is the smallest possible functional Snake game in C++ that runs in a terminal. It uses a single file, minimal dependencies, and relies on `std::cin` for input (no external libraries like ncurses).

### Key Features:
- **Terminal-based**: Uses standard ASCII characters.
- **Minimal code**: ~100 lines.
- **Input**: Arrow keys (or WASD).
- **Game Loop**: Runs until collision or game over.

### Code:

```cpp
#include <iostream>
#include <vector>
#include <random>
#include <chrono>
#include <thread>

using namespace std;

int main() {
    // Game state
    vector<pair<int, int>> snake = {{0, 0}}; // Head at (0,0)
    pair<int, int> food = {5, 5};
    int dx = 1, dy = 0; // Initial direction: right
    int score = 0;
    bool running = true;

    // Terminal dimensions
    int width = 20, height = 10;

    // Helper to clear screen
    auto clear = []() { cout << "\033[H"; };

    // Helper to print grid
    auto print = [&]() {
        clear();
        for (int y = 0; y < height; ++y) {
            for (int x = 0; x < width; ++x) {
                bool isSnake = false;
                for (auto& s : snake) {
                    if (s == {x, y}) {
                        isSnake = true;
                        break;
                    }
                }
                if (isSnake) cout << "█";
                else if ({x, y} == food) cout << "●";
                else cout << " ";
            }
            cout << "\n";
        }
        cout << "Score: " << score << "\n";
    };

    // Input handling
    auto handleInput = [&]() {
        char c;
        if (cin >> c) {
            if (c == 'w' || c == '↑') dy = -1, dx = 0;
            if (c == 's' || c == '↓') dy = 1, dx = 0;
            if (c == 'a' || c == '←') dy = 0, dx = -1;
            if (c == 'd' || c == '→') dy = 0, dx = 1;
        }
    };

    // Game loop
    while (running) {
        // Move snake
        pair<int, int> head = snake.front();
        pair<int, int> newHead = {head.first + dx, head.second + dy};

        // Check collision
        if (newHead.first < 0 || newHead.first >= width ||
            newHead.second < 0 || newHead.second >= height) {
            running = false;
            break;
        }

        // Check self-collision
        for (auto& s : snake) {
            if (s == newHead) {
                running = false;
                break;
            }
        }

        if (!running) break;

        // Move: add new head, remove tail unless food eaten
        snake.insert(snake.begin(), newHead);
        if (newHead == food) {
            score++;
            // Generate new food
            do {
                food = {rand() % width, rand() % height};
            } while (any_of(snake.begin(), snake.end(), [&](pair<int, int> s) {
                return s == food;
            }));
        } else {
            snake.pop_back();
        }

        print();
        handleInput();
        this_thread::sleep_for(chrono::milliseconds(100)); // Speed control
    }

    cout << "Game Over! Final Score: " << score << "\n";
    return 0;
}
```

### How to Run:
1. Save as `snake.cpp`.
2. Compile: `g++ snake.cpp -o snake`
3. Run: `./snake`

### Controls:
- **Arrow Keys** or **WASD** to move.
- **Space** or **Enter** to pause (optional, not implemented here for brevity).
- **Ctrl+C** to quit.

- **Minimalism**: Uses `std::vector` for the snake body, `std::cin` for input, and ANSI escape codes for clearing the screen.
