Here is the smallest possible functional Breakout game in a single HTML file. It uses the HTML5 Canvas API and requires no external libraries.

```html
<!DOCTYPE html>
<html>
<body>
<canvas id="c" width="200" height="200"></canvas>
<script>
const c = document.getElementById('c');
const ctx = c.getContext('2d');
let x = 100, y = 180, dx = 2, dy = -2;
let bx = 80, by = 190, bw = 40, bh = 10;
let bricks = [];
for(let r=0; r<5; r++) for(let col=0; col<10; col++) bricks.push({x: col*20, y: r*20, alive: true});

function loop() {
    ctx.clearRect(0, 0, 200, 200);
    ctx.fillStyle = '#000';
    ctx.fillRect(bx, by, bw, bh);
    
    // Move ball
    x += dx; y += dy;
    
    // Wall collision
    if(x < 0 || x > 200) dx = -dx;
    if(y < 0 || y > 200) dy = -dy;
    
    // Paddle collision
    if(y > by && x > bx && x < bx + bw) dy = -dy;
    
    // Brick collision
    bricks.forEach(b => {
        if(b.alive && x > b.x && x < b.x+20 && y > b.y && y < b.y+20) {
            b.alive = false; dy = -dy;
        }
    });
    
    // Draw bricks
    bricks.forEach(b => {
        if(b.alive) {
            ctx.fillStyle = '#f00';
            ctx.fillRect(b.x, b.y, 20, 20);
        }
    });
    
    requestAnimationFrame(loop);
}

document.addEventListener('keydown', e => {
    if(e.key === 'ArrowLeft') bx -= 10;
    if(e.key === 'ArrowRight') bx += 10;
});

loop();
</script>
</body
