4.  **Draft - Mental Code Structure:**
   ```html
   <!DOCTYPE html>
   <html>
   <head>
     <style>
       canvas { background: #000; display: block; margin: auto; }
     </style>
   </head>
   <body>
     <canvas id=c width=200 height=200></canvas>
     <script>
       const c=document.getElementById('c'), ctx=c.getContext('2d');
       const S=10, W=c.width/S, H=c.height/S;
       let snake=[{x:10,y:10}], dir={x:1,y:0}, food={x:15,y:15}, score=0;
       function draw(){
         ctx.fillStyle='#000'; ctx.fillRect(0,0,c.width,c.height);
         ctx.fillStyle='#0f0';
         for(let p of snake) ctx.fillRect(p.x*S,p.y*S,S,S);
         ctx.fillStyle='#f00'; ctx.fillRect(food.x*S,food.y*S,S,S);
       }
       function move(){
         let nx=snake[0].x+dir.x, ny=snake[0].y+dir.y;
         if(nx<0||nx>=W||ny<0||ny>=H||snake.some(p=>p.x===nx&&p.y===ny)) return;
         snake.unshift({x:nx,y:ny});
         if(nx===food.x&&ny===food.y){
           food={x:Math.floor(Math.random()*W),y:Math.floor(Math.random()*H)};
           score++;
         } else snake.pop();
       }
       document.onkeydown=e=>{
         if(e.key==='ArrowUp') dir={x:0,y:-1};
         if(e.key==='ArrowDown') dir={x:0,y:1};
         if(e.key==='ArrowLeft') dir={x:-1,y:0};
         if(e.key==='ArrowRight') dir={x:1,y:0};
       };
       setInterval(()=>{draw();move();},100);
     </script>
   </body>
   </html>
   ```
