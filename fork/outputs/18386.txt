```cpp
#include <fstream>   // for std::ifstream
#include <iostream>  // for std::cout / std::cerr

static void print_line_count() {
    std::ifstream file(__FILE__);
    if (!file.is_open()) {
        std::cerr << "Error: could not open " << __FILE__ << "\n";
        return;
    }

    size_t line_count = 0;
    std::string line;
    while (std::getline(file, line)) {
        ++line_count;
    }

    std::cout << "Line count of " << __FILE__ << ": " << line_count << std::endl;
}
```
