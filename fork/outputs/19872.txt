/* ---------- 7-bag randomizer ---------- */
function makeBag() {
  // Shuffle the 7 classic tetrominoes (indices 0..6) into a bag.
  const bag = [0,1,2,3,4,5,6];
  for (let i = bag.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [bag[i], bag[j]] = [bag[j], bag[i]];
  }
  return bag;
}
