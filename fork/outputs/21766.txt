// returns the current timestamp as a string, the format of the timestamp depends on the parameter
std::string common_timestamp_to_string(int64_t timestamp, const std::string & format) {
    std::time_t time = static_cast<std::time_t>(timestamp);
    std::tm     tm;

#if defined(_WIN32)
    gmtime_s(&tm, &time);
#else
    gmtime_r(&time, &tm);
#endif

    char buffer[80];
    strftime(buffer, sizeof(buffer), format.c_str(), &tm);
    return buffer;
}

// returns the current timestamp as a string, the format of the timestamp depends on the parameter
std::string common_time_to_string(const std::string & format) {
    return common_timestamp_to_string(std::time(nullptr), format);
}
