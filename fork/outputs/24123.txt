static bool has_format(const std::string & x) {
    static std::unordered_set<std::string> supported_formats;
    std::transform(FORMAT_RULES.begin(), FORMAT_RULES.end(), std::inserter(supported_formats, supported_formats.end()), [](const auto & entry) { return entry.first; });
    return supported_formats.find(x) != supported_formats.end();
}

}
