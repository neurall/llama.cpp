   ```python
   import csv
   from collections import defaultdict
   from datetime import datetime

   def aggregate_sales(csv_file):
       region_totals = defaultdict(float)
       month_totals = defaultdict(float)

       with open(csv_file, 'r') as f:
           reader = csv.DictReader(f)
           for row in reader:
               # Parse date to extract month
               try:
                   date_obj = datetime.strptime(row['date'], '%Y-%m-%d')
                   month_key = date_obj.strftime('%Y-%m')
               except ValueError:
                   continue # Skip malformed dates

               amount = float(row['amount'])
               region_totals[row['region']] += amount
               month_totals[month_key] += amount

       print("Total amount per region:")
       for region, total in sorted(region_totals.items()):
           print(f"{region}: {total:.2f}")

       print("\nTotal amount per month:")
       for month, total in sorted(month_totals.items()):
           print(f"{month}: {total:.2f}")

   if __name__ == "__main__":
       aggregate_sales("sales.csv")
   ```
