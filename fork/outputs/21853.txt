bool parse_cpu_mask(const std::string & mask, std::vector<int32_t> & cpumask) {
    if (mask.empty()) {
        return false;
    }
    cpumask.clear();
    for (size_t i = 0; i < mask.size(); i++) {
        char c = mask[i];
        if (c == '0' || c == '1') {
            cpumask.push_back(c == '1' ? 1 : 0);
        } else {
            return false;
        }
    }
    return true;
}
