```javascript
const c=document.getElementById('c'),x=c.getContext('2d');
const W=10,H=20,S=30;
let board=[],score=0,gameOver=false;
// Initialize board
for(let i=0;i<H;i++)board.push(new Array(W).fill(0));

// Pieces
const pieces=[
  [[1,1,1,1]], // I
  [[1,1],[1,1]], // O
  [[0,1,0],[1,1,1]], // T
  [[1,0,0],[1,1,1]], // J
  [[0,0,1],[1,1,1]], // L
  [[0,1,1],[1,1,0]], // S
  [[1,1,0],[0,1,1]]  // Z
];
const colors=['#0ff','#ff0','#f0f','#00f','#f80','#0f0','#f00'];

let piece,px,py,pieceIdx;

function newPiece(){
  pieceIdx=Math.floor(Math.random()*7);
  piece=pieces[pieceIdx].map(r=>[...r]);
  px=Math.floor((W-piece[0].length)/2);
  py=0;
}

function collide(){
  for(let r=0;r<piece.length;r++)
    for(let c=0;c<piece[r].length;c++)
      if(piece[r][c]){
        let nx=px+c,ny=py+r;
        if(nx<0||nx>=W||ny>=H)return true;
        if(ny>=0&&board[ny][nx])return true;
      }
  return false;
}

function merge(){
  for(let r=0;r<piece.length;r++)
    for(let c=0;c<piece[r].length;c++)
      if(piece[r][c]&&py+r>=0)
        board[py+r][px+c]=pieceIdx+1;
}

function clearLines(){
  for(let r=H-1;r>=0;r--){
    if(board[r].every(v=>v)){
      board.splice(r,1);
      board.unshift(new Array(W).fill(0));
      score+=100;
      r++;
    }
  }
}

function rotate(){
  let rotated=piece[0].map((_,i)=>piece.map(r=>r[i]).reverse());
  let old=piece;
  piece=rotated;
  if(collide())piece=old;
}

function draw(){
  x.fillStyle='#000';
  x.fillRect(0,0,c.width,c.height);
  // Draw board
  for(let r=0;r<H;r++)
    for(let c=0;c<W;c++)
      if(board[r][c]){
        x.fillStyle=colors[board[r][c]-1];
        x.fillRect(c*S,r*S,S-1,S-1);
      }
  // Draw piece
  if(piece){
    for(let r=0;r<piece.length;r++)
      for(let c=0;c<piece[r].length;c++)
        if(piece[r][c]){
          x.fillStyle=colors[pieceIdx];
          x.fillRect((px+c)*S,(py+r)*S,S-1,S-1);
        }
  }
  // Score
  x.fillStyle='#fff';
  x.font='20px Arial';
  x.fillText('Score: '+score,10,30);
  if(gameOver){
    x.fillStyle='#f00';
    x.font='30px Arial';
    x.fillText('GAME OVER',50,300);
  }
}

function drop(){
  if(gameOver)return;
  py++;
  if(collide()){
    py--;
    merge();
    clearLines();
    newPiece();
    if(collide())gameOver=true;
  }
}

document.addEventListener('keydown',e=>{
  if(gameOver)return;
  if(e.key==='ArrowLeft'){px--;if(collide())px++;}
  if(e.key==='ArrowRight'){px++;if(collide())px--;}
  if(e.key==='ArrowDown'){drop();}
  if(e.key==='ArrowUp'){rotate();}
});

newPiece();
draw();
setInterval(()=>{drop();draw();},500);
```
