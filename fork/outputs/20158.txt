### JavaScript Logic (game.js)

Now, let's add some basic JavaScript logic to control the snake and keep track of the score.

```javascript
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');

let snake = [{ x: 5, y: 5 }]; // Starting position and length
let food = { x: 10, y: 10 }; // Initial food position
let score = 0;

function update() {
    // Clear the canvas
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Move the snake
    snake[0].x += snake[1].x;
    snake[0].y += snake[1].y;

    // Check for collisions
    if (snake[0].x < 0 || snake[0].x >= canvas.width ||
        snake[0].y < 0 || snake[0].y >= canvas.height ||
        snake.length > 20) {
        clearTimeout(timeout);
        alert("Game Over!");
    } else {
        snake.unshift(snake[0]); // Add the head back to the end to form a new tail
    }

    // Draw the snake
    ctx.fillStyle = 'green';
    ctx.beginPath();
    ctx.moveTo(snake[0].x, snake[0].y);
    for (let i = 1; i < snake.length; i++) {
        ctx.lineTo(snake[i].x, snake[i].y);
    }
    ctx.closePath();
    ctx.fill();

    // Check for food and eat it
    if (snake[0].x === food.x && snake[0].y === food.y) {
        score += 1;
        food = { x: Math.floor(Math.random() * (canvas.width - 10)) + 10, y: Math.floor(Math.random() * (canvas.height - 10)) + 10 };
    }

    // Update the score
    document.getElementById('score').textContent = `Score: ${score}`;
}

let timeout = setTimeout(update, 100); // Update every 100 milliseconds

// Start the game
update();
```
