#include <iostream>
#include <vector>
#include <random>

using namespace std;

int main() {
    int w = 20, h = 15;
    vector<pair<int, int>> snake = {{10, 7}, {10, 8}, {10, 9}};
    pair<int, int> food = {rand() % w, rand() % h};
    int dx = 0, dy = -1; // Start moving up
    char c;

    while (true) {
        // Clear screen
        for (int y = 0; y < h; ++y) {
            for (int x = 0; x < w; ++x) {
                cout << (find(snake.begin(), snake.end(), {x, y}) != snake.end() ? '#' : 
                         (x == food.first && y == food.second ? 'F' : ' '));
            }
            cout << endl;
        }

        // Input
        if (cin >> c) {
            if (c == 'w') dx = 0, dy = -1;
            if (c == 's') dx = 0, dy = 1;
            if (c == 'a') dx = -1, dy = 0;
            if (c == 'd') dx = 1, dy = 0;
        }

        // Move head
        pair<int, int> head = {snake.front().first + dx, snake.front().second + dy};

        // Check collision
        if (head.first < 0 || head.first >= w || head.second < 0 || head.second >= h ||
            find(snake.begin(), snake.end(), head) != snake.end()) {
            cout << "Game Over!" << endl;
            break;
        }
