- Player (ship) that moves with arrow keys

- Score tracking
- Game over condition

```html
<!DOCTYPE html>
<html>
<body>
<canvas id=c width=400 height=400></canvas>
<script>
const ctx=c.getContext('2d'), p={x:200,y:350,s:10}, e=[], b=[], sc=0, k={};
document.onkeydown=e=>{k[e.key]=1};
document.onkeyup=e=>{k[e.key]=0};
document.onkeydown=e=>{if(e.key==' ')b.push({x:p.x,y:p.y});};
function loop(){
  ctx.fillStyle='#000';ctx.fillRect(0,0,400,400);
  if(k.ArrowLeft)p.x-=5;if(k.ArrowRight)p.x+=5;if(k.ArrowUp)p.y-=5;if(k.ArrowDown)p.y+=5;
  if(Math.random()<.05)e.push({x:Math.random()*400,y:0,s:8});
  for(let i=e.length-1;i>=0;i--){
    e[i].y+=2;
    if(e[i].y>400){e.splice(i,1);continue;}
    for(let j=b.length-1;j>=0;j--){
      if(Math.abs(b[j].x-e[i].x)<15&&Math.abs(b[j].y-e[i].y)<15){
        sc++;b.splice(j,1);e.splice(i,1);break;
      }
    }
    if(Math.abs(p.x-e[i].x)<15&&Math.abs(p.y-e[i].y)<15){
      ctx.fillStyle='#f00';ctx.fillText('GAME OVER',200,200);return;
    }
  }
  for(let i=b.length-1;i>=0;i--){
    b[i].y-=8;if(b[i].y<0)b.splice(i,1);
  }
  ctx.fillStyle='#0f0';ctx.fillRect(p.x-p.s,p.y-p.s,p.s*2,p.s*2);
  ctx.fillStyle='#fff';for(let i=b.length-1;i>=0;i--)ctx.fillRect(b[i].x-2,b[i].y-2,4,4);
  ctx.fillStyle='#f00';for(let i=e.length-1;i>=0;i--)ctx.fillRect(e[i].x-e[i].s,e[i].y-e[i].s,e[i].s*2,e[i].s*2);
  ctx.fillStyle='#fff';ctx.fillText('Score:'+sc,10,10);
  requestAnimationFrame(loop);
}
loop();
</script>
</body>
</html>
```

**Controls:**
- **Arrow keys** – move your ship (green square)
- **Space** – fire bullets (white dots)
- **Red squares** are enemies – shoot them to score
- Hit an enemy → score +1
- Enemy hits you → **Game Over**

~250 bytes of code, fully playable! 🚀🎯
